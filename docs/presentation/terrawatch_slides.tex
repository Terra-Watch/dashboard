% P8. SCENARIO ENGINE MATHS
% ============================================================
\begin{frame}{14. Scenario Engine: Elliptical Fire Growth}
\framesubtitle{Anderson length-to-breadth ratio with a Richards ellipse; ignition at the rear focus}
\scriptsize
\begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    With $U$ the mid-flame wind speed in m/s ($U = $ km/h $/\,3.6$):
    \[
    LB = 0.936\,e^{0.2566\,U} + 0.461\,e^{-0.1548\,U} - 0.397,\qquad LB \ge 1
    \]
    \[
    e = \frac{\sqrt{LB^{2}-1}}{LB},\qquad
    \text{head} = R\,t,\qquad
    \text{back} = \text{head}\,\frac{1-e}{1+e}
    \]
    \[
    a = \frac{\text{head}+\text{back}}{2},\qquad
    b = \frac{a}{LB},\qquad
    r(\varphi)=\frac{a(1-e^{2})}{1-e\cos\varphi}
    \]
    $R$ is the head-fire rate of spread (typed by the planner), $t$ the elapsed time, $\varphi$ the angle from the head direction (downwind).

    \column{0.46\textwidth}
    \textbf{\textcolor{amritaMaroon}{Uncertainty bands}}
    \begin{itemize}\setlength\itemsep{2pt}
        \item Wind speed $\pm$, direction $\pm$ and rate $\pm$ as fractions or degrees
        \item Outer band: radial maximum over the eight $\pm$ corners and the nominal
        \item Inner band: all-minus corner
        \item Exposure: earliest hour a place falls inside the outer / inner band
    \end{itemize}
    \textbf{\textcolor{amritaMaroon}{Not modelled}}\\
    Slope, spotting, crowning, fuel-moisture change, barriers.
\end{columns}
\end{frame}

\begin{frame}{16. Fire Evidence Assessment in the Workflow}
\framesubtitle{Advisory output, human review, no automatic status change}
\scriptsize
\begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \textbf{\textcolor{amritaMaroon}{How it is used}}
    \begin{itemize}\setlength\itemsep{2pt}
        \item Each evidence photo may carry an automated assessment
        \item Always labelled \emph{experimental, not for operational use}
        \item The queue never calls an assessment ``complete''
        \item The result is shown with its status; unavailable or pending states are explicit
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{\textcolor{amritaMaroon}{Human review}}
    \begin{itemize}\setlength\itemsep{2pt}
        \item The reviewer records a judgement of the \emph{photo}: genuine, may be fake or edited (reason required), seen with no conclusion, or cleared
        \item A review never changes the incident's status
        \item Inference is controlled by a server switch
    \end{itemize}
\end{columns}
\vspace{0.2cm}
\begin{center}
\fcolorbox{amritaMaroon}{white}{\parbox{0.9\textwidth}{\centering
\textbf{Rule:} an automated result informs a person; a person decides.}}
\end{center}
\end{frame}

% P11. PRIVACY AND SAFETY
% ============================================================
\begin{frame}{17. Safety and Privacy by Design}
\framesubtitle{Rules the product enforces, not guidelines}
\scriptsize
\begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \textbf{\textcolor{amritaMaroon}{Safety}}
    \begin{itemize}\setlength\itemsep{2pt}
        \item An empty map is never presented as ``no fire''
        \item Public map: verified incidents only, snapped to H3 cells and delayed
        \item Official notices need a second person's approval
        \item No emergency numbers, thresholds or retention periods are built in; they come from configuration, and a feature without its setting stays off and says so
        \item Community posts never alert responders and always point to ``Report''
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{\textcolor{amritaMaroon}{Privacy}}
    \begin{itemize}\setlength\itemsep{2pt}
        \item Citizens can download or delete their data; deleted accounts are anonymised
        \item Reports handed to staff stay as safety records without the name
        \item Phone numbers, links and coordinates in community posts wait for a moderator
        \item Evidence photos are returned with no-store caching
        \item Secrets are never printed or committed
    \end{itemize}
\end{columns}
\end{frame}

% P18. LIMITATIONS AND FUTURE WORK

\begin{frame}{24. Limitations and Future Work}
\scriptsize
\begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \textbf{\textcolor{amritaMaroon}{Limitations}}
    \begin{itemize}\setlength\itemsep{2pt}
        \item Forecast average precision is moderate in absolute terms; report it beside the event base rate
        \item Satellite detection time and place differ from the actual ignition
        \item The elliptical scenario model ignores slope, spotting and crowning, and is not validated against closed incidents yet
        \item Hindi and Odia text need native-speaker review
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{\textcolor{amritaMaroon}{Future work}}
    \begin{itemize}\setlength\itemsep{2pt}
        \item Validation page comparing saved scenarios with the final observed extent
        \item Link the occurrence forecast and the spread simulation to the Control Room screens
        \item Officer position sharing and lone-worker check-in, agreed with the department
        \item Field testing of the sensor units with Range Officers
    \end{itemize}
\end{columns}

% P19. CONCLUSION

\begin{frame}{25. Conclusion}
\scriptsize
\begin{itemize}\setlength\itemsep{5pt}
    \item \textbf{Forecasting:} a multimodal model that ranks forest cells by ignition risk and outperforms four baselines on both rolling-origin folds.
    \item \textbf{Platform:} one application where citizens report, the Control Room verifies and informs, and Range Officers respond, even offline.
    \item \textbf{Trust by design:} a confidence ladder, human decisions at every step, advisory-only automation, and server-enforced privacy.
    \item \textbf{Planning aid:} a scenario planner that draws uncertainty and never says what will burn.
    \item \textbf{Field data:} low-cost sensors to supplement satellite and weather inputs.
\end{itemize}
